cmake_minimum_required(VERSION 3.20)
project(deduplicating_file_store VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_library(dedup src/chunker.cpp src/io.cpp src/store.cpp)
target_include_directories(dedup PUBLIC include)
target_link_libraries(dedup PUBLIC OpenSSL::Crypto Threads::Threads)
target_compile_options(dedup PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dedup-store src/main.cpp)
target_link_libraries(dedup-store PRIVATE dedup)
option(DEDUP_BUILD_TESTS "Build GoogleTest specifications" ON)
if(DEDUP_BUILD_TESTS)
  include(FetchContent)
  FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.17.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(googletest)
  enable_testing()
  add_executable(dedup-tests tests/store_test.cpp)
  target_link_libraries(dedup-tests PRIVATE dedup GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(dedup-tests)
endif()
