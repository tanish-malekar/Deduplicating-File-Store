cmake_minimum_required(VERSION 3.20)
project(deduplicating_file_store VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_library(dedup src/chunker.cpp src/io.cpp src/store.cpp)
target_include_directories(dedup PUBLIC include)
target_link_libraries(dedup PUBLIC OpenSSL::Crypto Threads::Threads)
target_compile_options(dedup PRIVATE -Wall -Wextra -Wpedantic)
